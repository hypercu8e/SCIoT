cmake_minimum_required(VERSION 3.16)
project(sciot_firmware_host LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 11)
if(NOT TFLM_ROOT)
  message(FATAL_ERROR "Pass -DTFLM_ROOT=/path/to/TensorFlowLite_ESP32/src")
endif()
get_filename_component(REPO_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
set(LITE "${TFLM_ROOT}/tensorflow/lite")
file(GLOB MICRO_SOURCES "${LITE}/micro/*.cpp")
list(FILTER MICRO_SOURCES EXCLUDE REGEX "/(all_ops_resolver|test.*|mock.*|fake.*)\\.cpp$")
file(GLOB ALLOCATOR_SOURCES "${LITE}/micro/arena_allocator/*.cpp" "${LITE}/micro/memory_planner/*.cpp")
set(KERNEL_SOURCES "${LITE}/micro/kernels/kernel_util.cpp")
foreach(NAME add mul conv depthwise_conv pad softmax activations)
  list(APPEND KERNEL_SOURCES "${LITE}/micro/kernels/${NAME}.cpp")
  if(EXISTS "${LITE}/micro/kernels/${NAME}_common.cpp")
    list(APPEND KERNEL_SOURCES "${LITE}/micro/kernels/${NAME}_common.cpp")
  endif()
endforeach()
file(GLOB CORE_SOURCES "${LITE}/core/api/*.cpp")
option(SCIOT_SANITIZERS "Enable host memory checks" ON)
add_library(sciot_micro STATIC ${MICRO_SOURCES} ${ALLOCATOR_SOURCES}
  ${CORE_SOURCES} ${KERNEL_SOURCES} "${LITE}/c/common.cpp"
  "${LITE}/kernels/kernel_util.cpp" "${LITE}/kernels/internal/quantization_util.cpp"
  "${LITE}/kernels/internal/reference/portable_tensor_utils.cpp" "${LITE}/schema/schema_utils.cpp")
target_include_directories(sciot_micro PUBLIC "${REPO_ROOT}/src/client/common"
  "${TFLM_ROOT}" "${TFLM_ROOT}/third_party/flatbuffers/include"
  "${TFLM_ROOT}/third_party/gemmlowp" "${TFLM_ROOT}/third_party/ruy")
target_compile_definitions(sciot_micro PUBLIC TF_LITE_STATIC_MEMORY TF_LITE_DISABLE_X86_NEON)
target_compile_options(sciot_micro PUBLIC -fno-exceptions -ffp-contract=off -fno-fast-math -UNDEBUG)
if(SCIOT_SANITIZERS)
  target_compile_options(sciot_micro PUBLIC -O1 -g -fno-omit-frame-pointer -fsanitize=address,undefined)
  target_link_options(sciot_micro PUBLIC -fsanitize=address,undefined)
else()
  target_compile_options(sciot_micro PUBLIC -O3)
endif()
add_executable(firmware_host firmware_host.cpp)
target_link_libraries(firmware_host PRIVATE sciot_micro)
